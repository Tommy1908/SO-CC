int main() {
    struct res touch_data;
    struct screen draw_struct;

    bool en_movimiento = false;
    int start_x = 0, start_y = 0;

    int pantalla = open("/dev/pantalla");

    while(1) {
        // La lectura es bloqueante, el proceso duerme si no hay gestos
        read(pantalla,&touch_data);

        int current_x = getX(touch_data.TCH_COORD);
        int current_y = getY(touch_data.TCH_COORD);

        // Detección de inicio de gesto (tiempo 0)
        if (touch_data.TCH_TIME == 0) {
            en_movimiento = false;
            start_x = current_x;
            start_y = current_y;
        }

        // Si detectamos cambio de coordenadas, es un deslizamiento
        else if (current_x != start_x || current_y != start_y) {
            en_movimiento = true;

            // Deslizamiento: colorear de ROJO cada punto detectado
            draw_struct.x = current_x;
            draw_struct.y = current_y;
            draw_struct.color = ROJO;
            write(pantalla,&draw_struct);
        }

        // Fin del gesto: evaluamos si fue un punto fijo (Click o Prolongada)
        if (touch_data.TCH_STATUS == TCH_UP) {
            if (!en_movimiento) {
                draw_struct.x = start_x;
                draw_struct.y = start_y;

                // Click: duración menor a 500 ms -> AZUL
                if (touch_data.TCH_TIME < 500) {
                    draw_struct.color = AZUL;
                }
                // Pulsación prolongada: duración mayor al click -> VERDE
                else {
                    draw_struct.color = VERDE;
                }

                write(pantalla,&draw_struct);
            }
        }
    }
    close(pantalla);
    return 0;
}

Esto es una salida inesperada.

Va a escribir [despues] AAAA, BBBB, y valor CCCC
