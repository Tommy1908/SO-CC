```c
int count_files_with_extension(char *dir, char *extension, int min_size){
    int contador=0;
    struct Ext2FSInode *inode = inode_for_path(dir);

                if(!es_punto && !es_punto_punto){ // evita recursion infinita
                    int res = count_files_with_extension(dir + entry->name,extension, min_size); // Asumamos que creo un buffer block size, (una guasada pero asi aseguramos que nombre mas dir deberia entrar) y los concateno con un formateador de texto.                     contador += res;
                }
            }
            else if(entry->file_type == r && entry->inode != 0){ // r para regular

                if(strcmp(file_ext, extension)==0){ // misma extension

                        contador++;
                    }
                }
            }

            offset += entry->record_length;
        }

    }

    return contador;
}
```

En ext2, los directorios se almacenan en bloques completos. Dentro de cada bloque, las entradas se recorren mediante record_length, y la última entrada cubre el espacio restante. Por eso ambas condiciones son correctas.
