        char *cmd = malloc(tam_imprimir+5 * sizeof(char));
        if (cmd == NULL)
            exit(1);
        if (usuario_habilitado("/etc/shadow", nombre_usuario, clave)) {
            snprintf(cmd, tam_imprimir+4, "echo %s", imprimir);
            system(cmd);
        } else {
            printf("El usuario o clave indicados son incorrectos.");
            assert(-1);
    }
}
