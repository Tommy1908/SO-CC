volatile sig_atomic_t hijos_activos = 0;

void handlerSigchild(int sig){

	int saved = errno;

    while(waitpid(-1,NULL, WNOHANG) > 0){
        hijos_activos--;
    }
    errno = saved;
}

int main_servidor() {
    int N = 20;
    int servidor_fd;
    struct sockaddr_un direccion;
    socklen_t addrlen = sizeof(direccion);
    direccion.sun_family = AF_UNIX;
    strcpy(direccion.sun_path, "unix_socket");
    unlink(direccion.sun_path);
    servidor_fd = socket(AF_UNIX, SOCK_STREAM, 0);
    //...
    bind(servidor_fd, (struct sockaddr *)&direccion, sizeof(direccion));
    listen(servidor_fd, 64); // Cuantos puede haber maximo en backlog?

    signal(SIGCHLD, handlerSigchild);

    while(1){
        while(hijos_activos >= N){
            pause(); //Sigchild nos levanta
        }

        int client_fd;
        struct sockaddr_un client_addr;
        socklen_t addrlen = sizeof(client_addr);
        client_fd = accept(servidor_fd, (struct sockaddr *)&client_addr, &addrlen);

        if(client_fd < 0){
            if(errno==EINTR){
				continue; // SIGCHLD error, volvemos a aceptar
			}
			perror("error al aceptar");
			break;
        }

        hijos_activos++; // evitar race condition
		pid_t son = fork();
		if(son == 0){
			close(servidor_fd); // el hijo no tiene que saber de escuchar ninguno nuevo
			handle_client(client_fd);
			exit(EXIT_SUCCESS);
		}
		close(client_fd); // padre no necesita saber del cliente, se encarga su hijo
	}
	
	close(servidor_fd);
	return EXIT_SUCCESS;
}
